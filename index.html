🗄️ إدارة قاعدة البيانات

هذه الصفحة مخصصة لك لإدارة بنية قاعدة البيانات. شغّل الكودين في Supabase SQL Editor مرة واحدة فقط، وبعدها تعمل كل الجداول تلقائياً من التطبيق.

1️⃣ كود إنشاء الجداول (شغّله أولاً)

-- ===== إعداد الجداول (شغّله مرة واحدة في SQL Editor ثم Run) =====
create table if not exists classes (
  id text primary key,
  name text not null
);

create table if not exists students (
  id text primary key,
  name text not null,
  class_id text references classes(id) on delete cascade
);

create table if not exists behaviors (
  id text primary key,
  name text not null,
  emoji text default '⭐',
  points integer not null default 3
);

create table if not exists observers (
  id bigserial primary key,
  name text unique not null
);

create table if not exists submissions (
  id text primary key,
  student_id text references students(id) on delete cascade,
  class_id text references classes(id),
  behavior_id text references behaviors(id),
  behavior_name text,
  emoji text,
  points integer,
  observer text,
  note text,
  recorded_at timestamptz not null default now()
);

alter table classes enable row level security;
alter table students enable row level security;
alter table behaviors enable row level security;
alter table observers enable row level security;
alter table submissions enable row level security;

-- سياسات مفتوحة (كما كان التطبيق سابقاً) — يمكن تشديدها لاحقاً
do $ declare t text; begin
  foreach t in array array['classes','students','behaviors','observers','submissions'] loop
    execute format('drop policy if exists "%1$s_all" on %1$s', t);
    execute format('create policy "%1$s_all" on %1$s for all using (true) with check (true)', t);
  end loop;
end $;
